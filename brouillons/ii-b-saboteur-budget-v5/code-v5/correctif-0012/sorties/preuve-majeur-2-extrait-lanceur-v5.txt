  78  def exiger_accord(config: dict, environnement: dict, carte=None) -> str | None:
  79      """Accord cité pour toute machine payante : en mode réel, `go_id` de la configuration (forme vérifiée par
  80      `configuration_v5`) égal à `IIB_GO_ID` posé par le lanceur de session sur la machine ; sinon refus. Mode jouet : refusé sur
  81      une machine payante — `IIB_GO_ID` posé ou carte visible (contre-lecture, m7) ; `carte` : essais (sinon `carte_visible()`)."""
  82      if config["mode"] != "reel":
  83          if environnement.get("IIB_GO_ID") or (carte_visible() if carte is None else carte):
  84              raise GardeArret("mode jouet sur une machine payante (accord posé sur la machine, ou carte visible) : refus — une machine "
  85                               "payante ne joue qu'une configuration réelle sous accord cité")
  86          return None
  87      go = config.get("go_id")
  88      if not go or environnement.get("IIB_GO_ID") != go:
  89          raise GardeArret(f"accord cité absent ou différent (configuration : {go!r} ; machine : {environnement.get('IIB_GO_ID')!r}) : aucune "
  90                           "machine payante sans accord cité (refus)")
  91      return go
