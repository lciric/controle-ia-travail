#!/usr/bin/env python3
"""Lot 2 des jeux de T0.7, étape 1 : assemblage de J-C (version 3) par le script officiel, avec garde d'égalité — v1.

1. Contrôle des 32 remises retenues contre la liste de la mère (`liste-32.txt`, chemins et sha256).
2. `assembler` du script officiel (`scripts/verifier_jeux_synthetiques_t07_v2.py`, e53c9c27…), `--version v3`,
   `--jeux domaine-cout` (les remises ne portent que J-C), énoncés et annexes scellés du lot 1, réserve scellée,
   graine 20261105 ; sortie dans un dossier temporaire neuf.
3. Garde d'égalité : l'empreinte de `domaine-cout-v3.jsonl` doit égaler celle du jalon 32 (99b4f1b2…) ; inégalité : code 1.
4. R5 : la même garde, sur un artefact (même assemblage, graine 20261106), doit être inégale (code 1 attendu).
Usage : python -I -B assemblage_garde_jc_v3_v1.py <racine du dépôt> <liste-32.txt> <dossier temporaire> <journal à créer>
"""
from __future__ import annotations

import datetime as dt
import hashlib
import subprocess
import sys
from pathlib import Path

RACINE, LISTE, TMP, JOURNAL = (Path(a) for a in sys.argv[1:5])
ATTENDU = "99b4f1b25daa84f16b6c8c8dec06fede52ec7538f085d95b09e7e20bec627447"
SCRIPT = RACINE / "scripts/verifier_jeux_synthetiques_t07_v2.py"
SCRIPT_ATTENDU = "e53c9c2729cf17ac1b0f3c6fbf81169c1d568bbccb85fb1535981b4094341d22"
PY = RACINE / ".venv/bin/python"
ENTREES = {
    "donnees/t07-jeux-synthetiques-v2/enonces-reserve-v2.jsonl":
        "a9e8839f886f9df1ce8270a347151bd11bee0e0742b425edf5914cd5c23d50f8",
    "donnees/t07-jeux-synthetiques-v2/banque-instructions-v2.json":
        "230099ef0eece0c11c1ac1a2dfd563e5747292723ddd614eebe7bdbb37d1cc01",
    "donnees/t07-jeux-synthetiques-v2/banque-incitations-v2.json":
        "eb6a424d20505819491608e2b8c277735157801bb1230304b5f616922ff91b56",
    "donnees/t07-jeux-synthetiques-v2/mots-interdits-v2.json":
        "5d42ab633dd4dda055989a2609fc77e49accdb428f2acdbad0bdbaf9630cba76",
    "donnees/20261007-082440-t05-extraction/taches-entrainement.json":
        "8d20a698d76baa8a42cd1494227fc411ef043d24a67a413e5cf812541166a4fd",
}


def sha(p: Path) -> str:
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()


lignes: list[str] = []
fautes: list[str] = []


def j(t: str = "") -> None:
    lignes.append(t)
    print(t)


def maintenant() -> str:
    return dt.datetime.now(dt.timezone.utc).strftime("%Y-%m-%d %H:%M:%S UTC")


j("Journal d'assemblage de J-C, version 3 (lot 2 des jeux de T0.7), avec garde d'égalité — v1")
j(f"date : {maintenant()} (horloge de la machine, UTC)")
moi = Path(__file__).resolve()
j(f"outil : {moi.name} sha256 {sha(moi)} (texte intégral en fin de journal)")
j(f"script officiel : {SCRIPT.relative_to(RACINE)} sha256 {sha(SCRIPT)} (attendu {SCRIPT_ATTENDU})")
if sha(SCRIPT) != SCRIPT_ATTENDU:
    fautes.append("script officiel : empreinte inattendue")
for rel, att in ENTREES.items():
    e = sha(RACINE / rel)
    j(f"entrée : {rel} sha256 {e} ({'conforme' if e == att else 'NON CONFORME'})")
    if e != att:
        fautes.append(f"entrée {rel} : empreinte inattendue")
j(f"liste de la mère : {LISTE} sha256 {sha(LISTE)}")

remises: list[Path] = []
for ligne in LISTE.read_text(encoding="utf-8").splitlines():
    if not ligne.strip():
        continue
    att, chemin = ligne.split(None, 1)
    p = Path(chemin.strip())
    e = sha(p)
    remises.append(p)
    j(f"    remise {p} : {e} ({'OK' if e == att else 'DIFFÉRENTE'})")
    if e != att:
        fautes.append(f"remise {p} : empreinte ≠ liste")
if len(remises) != 32:
    fautes.append(f"{len(remises)} remises ≠ 32")
if fautes:
    j(f"ARRÊT avant assemblage : {fautes}")


def assembler(graine: int, sortie: Path) -> tuple[int, str]:
    cmd = [str(PY), "-I", "-B", str(SCRIPT), "assembler", *map(str, remises),
           "--annexes", "donnees/t07-jeux-synthetiques-v2",
           "--taches", "donnees/20261007-082440-t05-extraction/taches-entrainement.json",
           "--enonces", "donnees/t07-jeux-synthetiques-v2/enonces-reserve-v2.jsonl",
           "--graine", str(graine), "--sortie", str(sortie), "--version", "v3", "--jeux", "domaine-cout"]
    j("")
    j(f"== assembler, graine {graine} — début {maintenant()}")
    j("commande (depuis la racine du dépôt, PYTHONPATH=src) : " + " ".join(cmd[:5]) + " <32 remises de la liste, dans "
      "son ordre> " + " ".join(cmd[5 + len(remises):]))
    r = subprocess.run(cmd, cwd=RACINE, env={"PYTHONPATH": "src", "PATH": "/usr/bin:/bin"}, capture_output=True,
                       text=True)
    for t in (r.stdout + r.stderr).rstrip("\n").splitlines():
        j("    | " + t)
    j(f"code de sortie (assembler) : {r.returncode}")
    sortie_jc = sortie / "domaine-cout-v3.jsonl"
    return r.returncode, (sha(sortie_jc) if sortie_jc.exists() else "absent")


if not fautes:
    code, e = assembler(20261105, TMP / "assemblage-v3")
    egal = e == ATTENDU
    j(f"GARDE D'ÉGALITÉ : domaine-cout-v3.jsonl {e} ; attendu (jalon 32, contrôleur) {ATTENDU} : "
      f"{'ÉGALES' if egal else 'INÉGALES — arrêt du lot'}")
    if code != 0:
        fautes.append(f"assembler : code {code}")
    if not egal:
        fautes.append("garde d'égalité : inégale")
    code_a, e_a = assembler(20261106, TMP / "artefact-graine-20261106")
    egal_a = e_a == ATTENDU
    j(f"R5 — GARDE SUR L'ARTEFACT (graine 20261106) : {e_a} ; attendu {ATTENDU} : "
      f"{'ÉGALES — LA GARDE NE MORD PAS' if egal_a else 'INÉGALES, la garde refuse (code 1 attendu de la garde)'}")
    if egal_a:
        fautes.append("R5 : la garde n'a pas refusé l'artefact")

j("")
j(f"Bilan : {len(fautes)} faute(s) ; code {1 if fautes else 0}" + (f" — {fautes}" if fautes else ""))
j(f"fin : {maintenant()}")
j("")
j(f"--- texte intégral de l'outil ({moi.name}, sha256 {sha(moi)}) ---")
lignes.append(moi.read_text(encoding="utf-8").rstrip("\n"))
lignes.append("--- fin du texte de l'outil ---")
with open(JOURNAL, "x", encoding="utf-8", newline="\n") as f:
    f.write("\n".join(lignes) + "\n")
sys.exit(1 if fautes else 0)
--- fin du texte de l'outil ---
