def sha(p: Path) -> str:
    return hashlib.sha256(p.read_bytes()).hexdigest()

        enr = {"identifiant": t["identifiant"], "questions": t["questions"], "classification": t.get("classification")}

    j(f"    écrit : {cible.name} ({len(sortie_lignes)} lignes, clés identifiant, questions, classification ; "
      f"champs {', '.join(CHAMPS_CACHES)} jamais extraits) sha256 {sha(cible)}")
